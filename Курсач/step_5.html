Логика игры: Шаг 5 - Интерфейс (UI)

Игра не может обойтись без интерфейса! В нашем дереве сцен есть узел UI, содержащий текстовые метки (Labels) для вывода счета и сообщений (Старт, Пауза, Конец игры).

В функции _ready() (которая вызывается при старте игры) мы программно настраиваем их внешний вид: задаем цвет текста и добавляем тень для лучшей читаемости. А в функции new_game() мы управляем их видимостью.

func _ready():
    for node in $UI.get_children():
        if node is Label:
            node.add_theme_color_override("font_color", Color("#e0e0e0"))
            # Добавляем небольшую тень для читаемости
            node.add_theme_color_override("font_shadow_color", Color("#000000", 0.5))
            node.add_theme_constant_override("shadow_offset_x", 1)
            node.add_theme_constant_override("shadow_offset_y", 1)
    
    load_high_score()
    new_game()

func new_game():
    # ... (сброс позиций змейки) ...
    score = 0
    state = GameState.START
    
    # Управление интерфейсом
    $UI/GameOverLabel.visible = false
    $UI/StartLabel.visible = true
    $UI/PauseLabel.visible = false
    $UI/ScoreLabel.text = "Score: 0"
    $UI/HighScoreLabel.text = "Best: " + str(high_score)
